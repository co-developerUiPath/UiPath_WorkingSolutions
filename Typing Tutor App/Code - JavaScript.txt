const SFX = (() => {
  let ctx = null;
  function ensure(){
    if (!soundToggle.checked) return null;
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    if (!AudioCtx) return null;
    if (!ctx) ctx = new AudioCtx();
    return ctx;
  }
  function blip(kind){
    const c = ensure(); if (!c) return;
    const now = c.currentTime;
    const o = c.createOscillator();
    const g = c.createGain();
    const p = (kind==="err")
      ? {f0:180,f1:120,d:0.06,g:0.06}
      : (kind==="win")
      ? {f0:660,f1:880,d:0.08,g:0.05}
      : {f0:220,f1:180,d:0.035,g:0.04};

    o.type="triangle";
    o.frequency.setValueAtTime(p.f0, now);
    o.frequency.exponentialRampToValueAtTime(p.f1, now+p.d);
    g.gain.setValueAtTime(0.0001, now);
    g.gain.exponentialRampToValueAtTime(p.g, now+0.01);
    g.gain.exponentialRampToValueAtTime(0.0001, now+p.d);
    o.connect(g).connect(c.destination);
    o.start(now); o.stop(now+p.d+0.02);
  }
  return { blip };
})();

const clamp = (n,a,b)=>Math.max(a,Math.min(b,n));
const nowMs = ()=>performance.now();
const fmt = (sec)=>{
  sec = Math.max(0, Math.floor(sec));
  const m = Math.floor(sec/60), s = sec%60;
  return String(m).padStart(2,"0")+":"+String(s).padStart(2,"0");
};
const accPct = ()=> session.typed ? Math.round((session.correct/session.typed)*100) : 100;
const wpmVal = ()=>{
  const e = elapsedMs()/1000;
  if (e<=0) return 0;
  return Math.round(((session.correct/5)/e)*60);
};
const elapsedMs = ()=>{
  if (!session.startedAt) return 0;
  const base = (session.paused ? session.pausedAt : nowMs()) - session.startedAt;
  return Math.max(0, base - session.totalPausedMs);
};
const remainingSec = ()=>{
  if (session.kind!=="timed") return 0;
  return Math.max(0, session.preset - (elapsedMs()/1000));
};
const iso = ()=> new Date().toISOString();
function post(payload){
  try{ window.parent && window.parent.postMessage(payload,"*"); }catch(_){}
}
function esc(s){
  return String(s).replaceAll("&","&amp;").replaceAll("<","&lt;").replaceAll(">","&gt;");
}
const labelKey = (ch)=> ch===" " ? "Space" : ch==="\n" ? "Enter" : ch;

function getTargetWpm(mode){
  const d = clamp(difficulty[mode]||1,1,5);
  return CONFIG.targetWpmByDifficulty[d-1] || 35;
}
function pickText(mode){
  if (mode==="custom") return (customText.value||"").trim();
  const pack = CONFIG.packs[mode] || CONFIG.packs.email;
  const d = clamp(difficulty[mode]||1,1,5);
  return pack[d-1] || pack[0];
}
function buildParagraphText(mode, n){
  const pack = mode==="custom" ? null : (CONFIG.packs[mode] || CONFIG.packs.email);
  const base = clamp(difficulty[mode]||1,1,5);
  const parts = [];
  for (let i=0;i<n;i++){
    const d = clamp(base + (i>0?1:0), 1, 5);
    parts.push(mode==="custom" ? (customText.value||"").trim() : (pack[d-1] || pack[0]));
  }
  return parts.filter(Boolean).join("\n\n");
}

function setText(newText){
  session.text = (newText||"").replace(/\r\n/g,"\n").trim();
  if (!session.text) session.text = "Paste custom text, or choose a built-in mode to begin.";
  session.expected = Array.from(session.text);
  session.idx = 0;

  session.typed=0; session.correct=0; session.errors=0; session.streak=0;
  session.charRes = new Array(session.expected.length).fill(0);
  session.lastMistake=null;
  setToast(null);

  // Render spans
  const frag = document.createDocumentFragment();
  for (let i=0;i<session.expected.length;i++){
    const sp = document.createElement("span");
    sp.className="ch pending";
    sp.dataset.i = String(i);
    sp.textContent = session.expected[i];
    frag.appendChild(sp);
  }
  practiceTextEl.innerHTML="";
  practiceTextEl.appendChild(frag);
  paintCaret();
  updateKeyboardHighlight(false);
  updateHUD();
  scrollCaret();
  updateChips();
}

function paintCaret(){
  const kids = practiceTextEl.children;
  for (let i=0;i<kids.length;i++){
    kids[i].classList.toggle("caret", i===session.idx);
  }
}

  const el = practiceTextEl.querySelector(`.ch[data-i="${i}"]`);
  if (!el) return;
  el.classList.remove("pending","correct","incorrect");
  el.classList.add(cls);

function scrollCaret(){
  const el = practiceTextEl.querySelector(".ch.caret");
  if (!el) return;
  const cr = el.getBoundingClientRect();
  const sr = textScroller.getBoundingClientRect();
  if (cr.top < sr.top + 30 || cr.bottom > sr.bottom - 30){
    el.scrollIntoView({block:"center", behavior:"smooth"});
  }
}

function updateHUD(){
  const eSec = elapsedMs()/1000;
  const wpm = wpmVal();
  const acc = accPct();
  const left = remainingSec();
  const bestKey = `${session.mode}|${session.kind}`;
  const b = bestScores[bestKey];

  $("hud_wpm").textContent = session.active ? String(wpm) : "—";
  $("hud_acc").textContent = session.active ? `${acc}%` : "—";
  $("hud_err").textContent = session.active ? String(session.errors) : "—";
  $("hud_time").textContent = session.active
    ? (session.kind==="timed" ? `${fmt(session.preset-left)}/${fmt(session.preset)}` : fmt(eSec))
    : "—";
  $("hud_streak").textContent = session.active ? String(session.streak) : "—";
  $("hud_best").textContent = b ? `${b.wpm}wpm` : "—";

  const leftText = session.active ? (session.kind==="timed" ? `Time: ${fmt(left)}` : `Time: ${fmt(eSec)}`) : "Time: 00:00";
  timePillTop.textContent = leftText;
  timePillBottom.textContent = leftText;

  bestPill.textContent = b ? `Best: ${b.wpm} WPM • ${b.accuracy}%` : "Best: —";
  statusPill.textContent = session.active ? (session.paused ? "Paused" : "Active") : "Idle";
}

  for (let r=0;r<4;r++){

    row.dataset.cols = String(KEY_LAYOUT[r].length);

    keyboardEl.appendChild(row);
  }
  const row = document.createElement("div");
  row.className="kbdRow"; row.dataset.cols="10";
  const back = makeKey("Backspace"); back.classList.add("wide2");
  const space = makeKey("Space"); space.classList.add("space");
  const enter = makeKey("Enter"); enter.classList.add("wide2");
  row.appendChild(back); row.appendChild(space); row.appendChild(enter);
  keyboardEl.appendChild(row);
}

function makeKey(label){
  const b = document.createElement("button");
  b.type="button"; b.className="kbdKey";
  b.dataset.key = label;
  b.textContent = label;
  b.setAttribute("aria-label", `Key ${label}`);
  b.addEventListener("click", ()=>{
    focusCapture();
    if (label==="Backspace") handleBackspace();
    else if (label==="Enter") handleChar("\n");
    else if (label==="Space") handleChar(" ");
    else handleChar(label);
  });
  return b;
}

function updateKeyboardHighlight(warn){
  const next = session.expected[session.idx] ?? "";
  nextKeyPill.textContent = `Next: ${labelKey(next) || "—"}`;

  const keys = keyboardEl.querySelectorAll(".kbdKey");
  keys.forEach(k=>{k.dataset.hot="false";k.dataset.warn="false";});

  let target = next;
  if (target===" ") target="Space";
  if (target==="\n") target="Enter";
  if (target.length===1 && /[A-Z]/.test(target)) target=target.toLowerCase();

  keys.forEach(k=>{
    if (k.dataset.key===target){
      k.dataset.hot="true";
      if (warn) k.dataset.warn="true";
    }
  });
}

function focusCapture(){ try{ capture.focus(); }catch(_){} }

function loadText(){
  session.lastMistake = null;
  setToast(null);

  let t = "";
  if (session.kind==="paragraph") t = buildParagraphText(session.mode, session.preset);
  else t = pickText(session.mode);

  setText(t);
}

function start(){
  if (session.active) return;

  if (session.mode==="custom" && !(customText.value||"").trim()){
    setToast({expected:"Custom text", got:"empty"});
    overlay.classList.add("show");
    return;
  }

  session.active = true;
  session.paused = false;
  session.startedAt = nowMs();
  session.totalPausedMs = 0;
  session.lastLivePost = 0;

  controlPill.textContent = "Typing…";
  btnPause.textContent = "Pause";
  focusCapture();
  tick();
  updateHUD();
}

function pauseResume(){
  if (!session.active) return;
  if (!session.paused){
    session.paused = true;
    session.pausedAt = nowMs();
    controlPill.textContent = "Paused";
    btnPause.textContent = "Resume";
    cancelAnimationFrame(session.raf); session.raf=0;
  } else {
    session.paused = false;
    session.totalPausedMs += (nowMs() - session.pausedAt);
    controlPill.textContent = "Typing…";
    btnPause.textContent = "Pause";
    focusCapture();
    tick();
  }
  updateHUD();
}

function end(completed, reason){
  cancelAnimationFrame(session.raf); session.raf=0;

  const eSec = elapsedMs()/1000;
  const wpm = wpmVal();
  const acc = accPct();

  const key = `${session.mode}|${session.kind}`;
  const prev = bestScores[key];
  if (!prev || wpm>prev.wpm || (wpm===prev.wpm && acc>prev.accuracy)){
    bestScores[key] = { wpm, accuracy: acc };
  }

  if (session.mode!=="custom"){
    const target = getTargetWpm(session.mode);
    const promote = acc >= CONFIG.promoteIf.minAccuracy && wpm >= (target + CONFIG.promoteIf.minWpmOverTarget);
    if (promote) difficulty[session.mode] = clamp((difficulty[session.mode]||1)+1,1,CONFIG.difficultyMax);
    else if (acc < 86 && (difficulty[session.mode]||1)>1) difficulty[session.mode] = (difficulty[session.mode]||1)-1;
  }

  post({
    type:"typingStats",
    wpm, accuracy: acc, errors: session.errors,
    mode: MODES.find(m=>m.id===session.mode)?.name || session.mode,
    modeId: session.mode,
    duration: session.kind==="timed" ? session.preset : Math.round(eSec),
    sessionKind: session.kind,
    completed: !!completed,
    timestamp: iso(),
    reason: reason || ""
  });

  session.active=false; session.paused=false;
  session.startedAt=0; session.totalPausedMs=0;
  controlPill.textContent = completed ? "Session complete" : "Ready";
  btnPause.textContent="Pause";
  if (completed) SFX.blip("win");

  updateChips();
  updateHUD();
  updateKeyboardHighlight(false);
}

function finishIfNeeded(){
  if (!session.active || session.paused) return;

  if (session.kind==="timed"){
    if (remainingSec() <= 0) end(true, "time");
  } else {
    if (session.idx >= session.expected.length) end(true, "paragraphComplete");
  }
}

function handleBackspace(){
  if (!session.active || session.paused) return;
  if (session.idx <= 0) return;

  session.idx -= 1;
  session.typed = Math.max(0, session.typed - 1);

  const prev = session.charRes[session.idx];
  if (prev === 1) session.correct = Math.max(0, session.correct - 1);
  if (prev === -1) session.errors = Math.max(0, session.errors - 1);

  session.charRes[session.idx] = 0;
  mark(session.idx, "pending");
  session.streak = 0;

  setToast(null);
  paintCaret();
  updateKeyboardHighlight(false);
  updateHUD();
  scrollCaret();
  SFX.blip("key");
}

function recordWeak(mode, expected){
  weakChars[mode][expected] = (weakChars[mode][expected] || 0) + 1;
}

function handleChar(ch){
  if (!session.active || session.paused) return;
  if (session.idx >= session.expected.length) return;

  const expected = session.expected[session.idx];
  const strict = strictToggle.checked;
  const ok = (ch === expected);

  session.typed += 1;

  if (ok){
    session.correct += 1;
    session.streak += 1;
    session.charRes[session.idx] = 1;
    mark(session.idx, "correct");
    setToast(null);
    SFX.blip("key");
    session.idx += 1;
    paintCaret();
    updateKeyboardHighlight(false);
    scrollCaret();
  } else {
    session.errors += 1;
    session.streak = 0;
    session.charRes[session.idx] = -1;
    mark(session.idx, "incorrect");
    recordWeak(session.mode, expected);
    setToast({expected, got: ch});
    updateKeyboardHighlight(true);
    SFX.blip("err");

    if (!strict){
      session.idx += 1;
      paintCaret();
      updateKeyboardHighlight(false);
      scrollCaret();
    }
  }

  updateHUD();
  finishIfNeeded();
}

function skip(){
  if (!session.active || session.paused) return;

  session.errors += CONFIG.skipPenaltyErrors;
  session.streak = 0;
  setToast({expected:"Current word", got:"skipped"});

  while (session.idx < session.expected.length){
    const e = session.expected[session.idx];
    mark(session.idx, "incorrect");
    recordWeak(session.mode, e);
    session.charRes[session.idx] = -1;
    session.idx += 1;
    if (e===" " || e==="\n" || e==="\t") break;
  }
  paintCaret();
  updateKeyboardHighlight(true);
  updateHUD();
  scrollCaret();
  finishIfNeeded();
}

function tick(){
  if (!session.active || session.paused) return;

  if (livePost.checked){
    const t = nowMs();
    if (!session.lastLivePost || (t - session.lastLivePost) >= CONFIG.liveUpdateEveryMs){
      session.lastLivePost = t;
      post({
        type:"typingLive",
        wpm: wpmVal(),
        accuracy: accPct(),
        errors: session.errors,
        timeLeft: session.kind==="timed" ? Math.ceil(remainingSec()) : 0,
        streak: session.streak,
        mode: MODES.find(m=>m.id===session.mode)?.name || session.mode,
        modeId: session.mode,
        timestamp: iso()
      });
    }
  }

  updateHUD();
  finishIfNeeded();
  session.raf = requestAnimationFrame(tick);
}

  session.mode = modeId;
  customBox.style.display = (modeId==="custom") ? "block" : "none";
  refreshModeButtons();
  updateChips();
  loadText();
  focusCapture();
}

function initPresets(){
  function fillTimed(){
    presetSel.innerHTML="";
    for (const s of CONFIG.durationPresetsSec){
      const o=document.createElement("option");
      o.value=String(s); o.textContent=`${s} seconds`;
      presetSel.appendChild(o);
    }
    presetSel.value=String(CONFIG.durationPresetsSec[1]||60);
  }
  function fillPara(){
    presetSel.innerHTML="";
    for (const n of CONFIG.paragraphPresets){
      const o=document.createElement("option");
      o.value=String(n); o.textContent=`${n} paragraph${n===1?"":"s"}`;
      presetSel.appendChild(o);
    }
    presetSel.value=String(CONFIG.paragraphPresets[0]||1);
  }

  fillTimed();
  session.kind = sessionKindSel.value;
  session.preset = Number(presetSel.value);

  sessionKindSel.addEventListener("change", ()=>{
    session.kind = sessionKindSel.value;
    if (session.kind==="timed") fillTimed(); else fillPara();
    session.preset = Number(presetSel.value);
    updateChips();
    loadText();
  });

  presetSel.addEventListener("change", ()=>{
    session.preset = Number(presetSel.value);
    updateChips();
    loadText();
  });
}

/* =========================
   Event wiring
   ========================= */
function wire(){
  btnStart.addEventListener("click", start);
  btnPause.addEventListener("click", pauseResume);
  btnRestart.addEventListener("click", ()=>{ end(false,"restart"); loadText(); });
  btnSkip.addEventListener("click", skip);
  btnNextText.addEventListener("click", ()=>{ end(false,"nextText"); loadText(); });

  btnTheme.addEventListener("click", ()=>{
    document.body.dataset.theme = (document.body.dataset.theme==="light") ? "dark" : "light";
  });
  btnFocus.addEventListener("click", focusCapture);

  textViewport.addEventListener("click", focusCapture);
  btnShowOnboarding.addEventListener("click", ()=>overlay.classList.add("show"));
  btnCloseOverlay.addEventListener("click", ()=>overlay.classList.remove("show"));
  btnEnterApp.addEventListener("click", ()=>{ overlay.classList.remove("show"); focusCapture(); });

  btnUseCustom.addEventListener("click", ()=>{
    if (session.mode==="custom"){ loadText(); focusCapture(); }
  });

  // hardware keyboard
  window.addEventListener("keydown", (e)=>{
    if (!session.active || session.paused) return;

    if (e.key===" " || e.key==="Spacebar") e.preventDefault();

    if (e.key==="Backspace"){ e.preventDefault(); handleBackspace(); return; }
    if (e.key==="Enter"){ e.preventDefault(); handleChar("\n"); return; }
    if (e.key.length===1){ handleChar(e.key); return; }

    if (e.key==="Escape"){ pauseResume(); }
  }, { passive:false });

  overlay.addEventListener("click",(e)=>{ if (e.target===overlay) overlay.classList.remove("show"); });

  document.addEventListener("pointerdown",(e)=>{
    const tag = (e.target && e.target.tagName) ? e.target.tagName.toLowerCase() : "";
    if (["select","option","textarea","input","button"].includes(tag)) return;
    if (session.active && !session.paused) focusCapture();
  });
}

function init(){
  initHUD();
  initKeyboard();
  initModes();
  initPresets();
  setMode("email");
  wire();

  overlay.classList.add("show");
  controlPill.textContent="Ready";
  updateHUD();
  focusCapture();
}
